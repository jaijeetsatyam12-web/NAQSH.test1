const AppShell: React.FC<{
  lang: Language;
  setLang: (lang: Language) => void;
}> = ({ lang, setLang }) => {
  const location = useLocation();

  // Scroll to top on route change
  useEffect(() => {
    window.scrollTo({ top: 0, left: 0, behavior: 'instant' as ScrollBehavior });
  }, [location.pathname]);

  // Sync <html lang>, <title>, and <meta name="description"> per route
  useEffect(() => {
    document.documentElement.lang = lang;

    const path = location.pathname;
    let seoEntry = PAGE_SEO[path];

    if (!seoEntry && path.startsWith('/departments/')) {
      const deptId = path.replace('/departments/', '');
      const dept = hospitalData.departments.find((d) => d.id === deptId);
      if (dept) {
        seoEntry = {
          title: {
            en: `${dept.name.en} | Naqsh Multi-Speciality Hospital, Patna`,
            hi: `${dept.name.hi} | नक़्श मल्टी-स्पेशलिटी हॉस्पिटल, पटना`,
          },
          description: dept.overview,
        };
      }
    }

    if (!seoEntry) {
      seoEntry = {
        title: {
          en: 'Page Not Found | Naqsh Multi-Speciality Hospital',
          hi: 'पेज नहीं मिला | नक़्श मल्टी-स्पेशलिटी हॉस्पिटल',
        },
        description: PAGE_SEO['/'].description,
      };
    }

    document.title = seoEntry.title[lang];
    const metaDesc = document.querySelector('meta[name="description"]');
    if (metaDesc) {
      metaDesc.setAttribute('content', seoEntry.description[lang]);
    }
    const ogTitle = document.querySelector('meta[property="og:title"]');
    if (ogTitle) {
      ogTitle.setAttribute('content', seoEntry.title[lang]);
    }
    const ogDesc = document.querySelector('meta[property="og:description"]');
    if (ogDesc) {
      ogDesc.setAttribute('content', seoEntry.description[lang]);
    }
  }, [location.pathname, lang]);

  // Scroll-reveal observer for .reveal-up elements
  useEffect(() => {
    const elements = document.querySelectorAll('.reveal-up');
    if (typeof IntersectionObserver === 'undefined') {
      elements.forEach((el) => el.classList.add('is-visible'));
      return;
    }

    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {
            entry.target.classList.add('is-visible');
            observer.unobserve(entry.target);
          }
        });
      },
      { threshold: 0.08, rootMargin: '0px 0px -24px 0px' }
    );

    elements.forEach((el) => observer.observe(el));
    return () => observer.disconnect();
  }, [location.pathname, lang]);

  const isBookingRoute =
    location.pathname === '/book' || location.pathname.startsWith('/book/');

  return (
    <div
      className={`min-h-screen flex flex-col bg-white text-[#475569] ${
        !isBookingRoute ? 'pb-20' : ''
      }`}
    >
      {/* 1. Sticky Navbar */}
      <Header lang={lang} setLang={setLang} />

      {/* Main Route Content with 150ms fade transition */}
      <main className="flex-1">
        <Suspense fallback={<RouteSkeleton />}>
          <div key={location.pathname} className="route-fade">
            <Routes>
              <Route path="/" element={<HomePage lang={lang} />} />
              <Route path="/about" element={<AboutPage lang={lang} />} />
              <Route
                path="/departments"
                element={<DepartmentsPage lang={lang} />}
              />
              <Route
                path="/departments/:id"
                element={<DepartmentDetailPage lang={lang} />}
              />
              <Route path="/doctors" element={<DoctorsPage lang={lang} />} />
              <Route
                path="/book"
                element={<BookAppointmentPage lang={lang} />}
              />
              <Route path="/gallery" element={<GalleryPage lang={lang} />} />
              <Route path="/contact" element={<ContactPage lang={lang} />} />
              <Route path="*" element={<NotFoundPage lang={lang} />} />
            </Routes>
          </div>
        </Suspense>
      </main>

      {/* 8. Footer */}
      <Footer lang={lang} />

      {/* Mobile Sticky Bottom Bar: Call | Book | WhatsApp (hidden on /book and desktop) */}
      <MobileBottomBar lang={lang} />
    </div>
  );
};

export function App() {
  const [lang, setLangState] = useState<Language>(() => {
    try {
      const saved = window.localStorage.getItem(LANG_STORAGE_KEY);
      if (saved === 'en' || saved === 'hi') {
        return saved;
      }
    } catch {
      // Fallback to English if localStorage is blocked
    }
    return 'en';
  });

  const [dragOffsetY, setDragOffsetY] = useState(0);
  const touchStartYRef = useRef<number | null>(null);

  slotLabel,

          </span>
          {slotLabel && (
            <span className="text-xs text-[#475569] mt-1 text-center max-w-xs">
              {slotLabel}
            </span>
          )}
        </div>

import { MIST_REST_START } from '../config.js';

export const VERTEX_SHADER_SOURCE = `#version 300 es
void main(){vec2 p=vec2((gl_VertexID<<1)&2,gl_VertexID&2);gl_Position=vec4(p*2.-1.,0,1);}`;

const FRAGMENT_TEMPLATE = `#version 300 es
#define OCT __OCT__
precision highp float;
uniform vec2 R;
uniform float T;
uniform float S;
out vec4 O;
float h(vec2 p){p=fract(p*vec2(123.34,456.21));p+=dot(p,p+45.32);return fract(p.x*p.y);}
float n(vec2 p){vec2 i=floor(p),f=fract(p);f=f*f*(3.-2.*f);return mix(mix(h(i),h(i+vec2(1,0)),f.x),mix(h(i+vec2(0,1)),h(i+vec2(1,1)),f.x),f.y);}
float fbm(vec2 p){float a=.5,s=0.;mat2 m=mat2(1.6,1.2,-1.2,1.6);p=mat2(.8,.6,-.6,.8)*p+5.3;for(int i=0;i<OCT;i++){s+=a*n(p);p=m*p;a*=.5;}return s;}
void main(){
vec2 uv=gl_FragCoord.xy/R; uv.y=1.-uv.y; vec2 p=vec2(uv.x*R.x/R.y,uv.y); float t=T*.04;
vec2 w=vec2(fbm(p*1.3+vec2(t,-t*.6)),fbm(p*1.3+vec2(-t*.8,t)+7.));
float d=fbm(p*1.8+w*1.6+vec2(t*1.5,-t*.9));
float low=smoothstep(MIST_REST_START,1.02,uv.y+(d-.5)*.36);
float band=smoothstep(.35,.7,d)*smoothstep(.45,.7,uv.y)*(1.-smoothstep(.8,1.,uv.y))*.30;
float rest=clamp(low*.92+band,0.,1.);
float front=1.05-S*.85;
float rise=smoothstep(front-.02,front+.28,uv.y+(d-.5)*.5);
float a=clamp(max(rest,max(rise*.96,S*(.25+.45*d))),0.,1.);
vec3 c=mix(vec3(.80,.90,.91),vec3(.99,1.,1.),clamp(d*1.2,0.,1.));
c=mix(c,vec3(.957,.984,.984),smoothstep(.88,1.,uv.y));
O=vec4(c*a,a);
}`;

export function buildMistFragmentShader(octaves = 4) {
  const safeRestStart = Number(MIST_REST_START).toFixed(2);
  return FRAGMENT_TEMPLATE.replace('__OCT__', String(octaves)).replace(
    'MIST_REST_START',
    safeRestStart
  );
}

export function createMistProgram(gl, octaves = 4) {
  const compile = (type, src) => {
    const shader = gl.createShader(type);
    if (!shader) return null;
    gl.shaderSource(shader, src);
    gl.compileShader(shader);
    if (!gl.getShaderParameter(shader, gl.COMPILE_STATUS)) {
      gl.deleteShader(shader);
      return null;
    }
    return shader;
  };

  const vs = compile(gl.VERTEX_SHADER, VERTEX_SHADER_SOURCE);
  const fs = compile(gl.FRAGMENT_SHADER, buildMistFragmentShader(octaves));
  if (!vs || !fs) {
    if (vs) gl.deleteShader(vs);
    if (fs) gl.deleteShader(fs);
    return null;
  }

  const prog = gl.createProgram();
  if (!prog) {
    gl.deleteShader(vs);
    gl.deleteShader(fs);
    return null;
  }

  gl.attachShader(prog, vs);
  gl.attachShader(prog, fs);
  gl.linkProgram(prog);
  gl.deleteShader(vs);
  gl.deleteShader(fs);

  if (!gl.getProgramParameter(prog, gl.LINK_STATUS)) {
    gl.deleteProgram(prog);
    return null;
  }

  const vao = gl.createVertexArray();

  return {
    program: prog,
    vao,
    uR: gl.getUniformLocation(prog, 'R'),
    uT: gl.getUniformLocation(prog, 'T'),
    uS: gl.getUniformLocation(prog, 'S'),
    dispose() {
      if (vao) gl.deleteVertexArray(vao);
      gl.deleteProgram(prog);
    },
  };
}

    const syncPhotoGeometry = () => {
      if (unmounted) return;
      heroHeight = heroEl.offsetHeight || 600;
      const cw = photoStageEl.clientWidth || window.innerWidth;
      const ch = photoStageEl.clientHeight || 360;
      const scale = Math.max(cw / IMG_NAT_W, ch / IMG_NAT_H);
      const bw = Math.ceil(IMG_NAT_W * scale);
      const bh = Math.ceil(IMG_NAT_H * scale);
      const bx = Math.round((cw - bw) * fx);

      // Anchor vertical focus so both the red NAQSH roof sign and all 11 rows of the blue board remain inside the unmasked zone
      let by = Math.round((ch - bh) * Math.max(fy, 0.52));
      const minBoardBottom = ch * 0.87;
      if (bh * 0.84 + by > minBoardBottom) {
        by = Math.round(minBoardBottom - bh * 0.84);
      }
      const maxSignTop = ch * 0.05;
      if (bh * 0.11 + by < maxSignTop) {
        by = Math.round(maxSignTop - bh * 0.11);
      }

      photoCoordEl.style.width = `${bw}px`;
      photoCoordEl.style.height = `${bh}px`;
      photoCoordEl.style.transform = `translate3d(${bx}px, ${by}px, 0)`;
    };

    syncPhotoGeometry();

    // Draw photo once into 40px-wide offscreen sky canvas after img.decode()
    const drawSkyCanvas = () => {
      if (unmounted || !skyCanvas || !imgEl) return;
      try {
        const w = 40;
        const h = Math.max(
          24,
          Math.round((40 * IMG_NAT_H) / IMG_NAT_W)
        );
        if (skyCanvas.width !== w) skyCanvas.width = w;
        if (skyCanvas.height !== h) skyCanvas.height = h;
        const ctx = skyCanvas.getContext('2d', { alpha: false });
        if (ctx) {
          ctx.drawImage(imgEl, 0, 0, w, h);
        }
      } catch {
        // Ignore if image not yet ready
      }
    };

    // WebGL2 setup & fallback state
    let gl = null;
    let mistProg = null;
    let useFallback = false;
    let currentOctaves = window.innerWidth >= 821 ? 5 : 4;
    let widthScale = 1.0;

    const activateFallback = () => {
      if (useFallback) return;
      useFallback = true;
      heroEl.classList.add('use-mist-fallback');
      if (mistProg) {
        mistProg.dispose();
        mistProg = null;
      }
      if (gl) {
        try {
          gl.getExtension('WEBGL_lose_context')?.loseContext();
        } catch {
          // Ignore
        }
        gl = null;
      }
    };

    const syncMistCanvasSize = () => {
      if (!mistCanvas || useFallback) return;
      const isDesktop = window.innerWidth >= 821;
      const baseW = isDesktop ? 640 : Math.min(window.innerWidth, 360);
      const targetW = Math.max(220, Math.round(baseW * widthScale));
      const stageW = photoStageEl.clientWidth || window.innerWidth;
      const stageH = (photoStageEl.clientHeight || 360) * 0.55;
      const aspect = stageH / Math.max(1, stageW);
      const targetH = Math.max(110, Math.round(targetW * aspect));

      if (mistCanvas.width !== targetW || mistCanvas.height !== targetH) {
        mistCanvas.width = targetW;
        mistCanvas.height = targetH;
      }
    };

    if (mistCanvas) {
      try {
        gl = mistCanvas.getContext('webgl2', {
          alpha: true,
          premultipliedAlpha: true,
          antialias: false,
          depth: false,
          stencil: false,
          powerPreference: 'low-power',
        });
        if (!gl) {
          activateFallback();
        } else {
          syncMistCanvasSize();
          mistProg = createMistProgram(gl, currentOctaves);
          if (!mistProg) {
            activateFallback();
          }
        }
      } catch {
        activateFallback();
      }
    } else {
      activateFallback();
    }

    const onContextLost = (e) => {
      e.preventDefault();
      activateFallback();
    };
    mistCanvas?.addEventListener('webglcontextlost', onContextLost, false);

    const renderMistFrame = (timeSec, surgeVal) => {
      if (useFallback || !gl || !mistProg || !mistCanvas) {
        if (fallbackMistRef.current) {
          const boost = Math.min(1, 0.78 + surgeVal * 0.22);
          fallbackMistRef.current.style.opacity = boost.toFixed(3);
        }
        return;
      }
      const w = mistCanvas.width;
      const h = mistCanvas.height;
      gl.viewport(0, 0, w, h);
      gl.useProgram(mistProg.program);
      if (mistProg.vao) gl.bindVertexArray(mistProg.vao);
      gl.uniform2f(mistProg.uR, w, h);
      gl.uniform1f(mistProg.uT, timeSec);
      gl.uniform1f(mistProg.uS, surgeVal);
      gl.drawArrays(gl.TRIANGLES, 0, 3);
    };

    // Clock & animation state (NEVER stored in React state)
    let rafId = null;
    let isInView = true;
    let isPaused = false;
    let simTimeSec = 0;
    let lastRafNow = performance.now();
    let frameTick = 0;

    // Intro state (1400ms on first load after img.decode())
    let isIntroActive = false;
    let introElapsedMs = 0;
    const INTRO_DURATION_MS = 1400;

    // Tap puff state (350ms rise to .45, 900ms return to 0)
    let isPuffActive = false;
    let puffElapsedMs = 0;
    const PUFF_RISE_MS = 350;
    const PUFF_FALL_MS = 900;
    const PUFF_TOTAL_MS = PUFF_RISE_MS + PUFF_FALL_MS;

    // Scroll parallax state
    let scrollDirty = true;

    // Adaptive quality tracking over drawn frames
    let drawnFrameCount = 0;
    let accumulatedFrameMs = 0;
    let qualityStage = 0; // 0 = initial, 1 = downgraded once, 2 = done

    const applyScrollParallax = () => {
      const sy = window.scrollY || window.pageYOffset || 0;
      const clamped = Math.min(Math.max(0, sy), heroHeight);
      const photoPar = clamped * 0.08;
      const mistPar = sy * 0.14;
      heroEl.style.setProperty('--par', `${photoPar.toFixed(2)}px`);
      if (mistWrapEl) {
        mistWrapEl.style.transform = `translate3d(0, ${mistPar.toFixed(2)}px, 0)`;
      }
      scrollDirty = false;
    };

    // Set initial CSS variables before decode completes
    if (isReducedMotion) {
      heroEl.classList.add('hero-reduced-motion');
      heroEl.style.setProperty('--intro', '1');
      heroEl.style.setProperty('--par', '0px');
    } else if (!hasPlayedHeroIntro) {
      heroEl.style.setProperty('--intro', '0');
      heroEl.style.setProperty('--par', '0px');
      heroEl.classList.add('is-intro-running');
      if (bloomEl) {
        bloomEl.style.opacity = '0';
        bloomEl.style.animation = 'none';
      }
      renderMistFrame(0, 1.0);
    } else {
      heroEl.style.setProperty('--intro', '1');
      heroEl.style.setProperty('--par', '0px');
    }

    const tick = (now) => {
      if (unmounted || isPaused) {
        rafId = null;
        return;
      }

      const rawDtMs = Math.max(0, Math.min(100, now - lastRafNow));
      lastRafNow = now;
      frameTick++;

      if (scrollDirty) {
        applyScrollParallax();
      }

      if (isReducedMotion) {
        renderMistFrame(0, 0);
        rafId = null;
        return;
      }

      simTimeSec += rawDtMs * 0.001;
      let surgeS = 0;
      const highRate = isIntroActive || isPuffActive;

      if (isIntroActive) {
        introElapsedMs += rawDtMs;
        const p = Math.min(1, introElapsedMs / INTRO_DURATION_MS);
        // Ease-out for mist clearing (1 -> 0)
        const inv = 1 - p;
        surgeS = inv * inv;
        // Ease-out cubic for photo rise & fade (0 -> 1)
        const introEase = 1 - inv * inv * inv;
        heroEl.style.setProperty('--intro', introEase.toFixed(4));

        // Bloom fades in during the last 600ms (800ms -> 1400ms)
        if (bloomEl) {
          if (introElapsedMs < 800) {
            bloomEl.style.opacity = '0';
          } else {
            const bp = Math.min(1, (introElapsedMs - 800) / 600);
            bloomEl.style.opacity = (bp * 0.72).toFixed(3);
          }
        }

        if (introElapsedMs >= INTRO_DURATION_MS) {
          isIntroActive = false;
          surgeS = 0;
          heroEl.style.setProperty('--intro', '1');
          heroEl.classList.remove('is-intro-running');
          if (bloomEl) {
            bloomEl.style.opacity = '';
            bloomEl.style.animation = '';
          }
        }
      } else if (isPuffActive) {
        puffElapsedMs += rawDtMs;
        if (puffElapsedMs <= PUFF_RISE_MS) {
          const pr = puffElapsedMs / PUFF_RISE_MS;
          const easeOut = 1 - (1 - pr) * (1 - pr);
          surgeS = 0.45 * easeOut;
          if (bloomEl) {
            const b = 0.72 + (0.95 - 0.72) * easeOut;
            bloomEl.style.opacity = b.toFixed(3);
          }
        } else if (puffElapsedMs < PUFF_TOTAL_MS) {
          const pf = (puffElapsedMs - PUFF_RISE_MS) / PUFF_FALL_MS;
          // Smooth ease-in-out return to 0
          const easeInOut = 0.5 * (1 + Math.cos(Math.PI * pf));
          surgeS = 0.45 * easeInOut;
          if (bloomEl) {
            const b = 0.72 + (0.95 - 0.72) * easeInOut;
            bloomEl.style.opacity = b.toFixed(3);
          }
        } else {
          isPuffActive = false;
          puffElapsedMs = 0;
          surgeS = 0;
          if (bloomEl) {
            bloomEl.style.opacity = '';
          }
        }
      }

      // Idle draws every 2nd frame (~30fps); intro and puff draw every frame
      const shouldDraw = highRate || (frameTick & 1) === 0;

      if (shouldDraw) {
        renderMistFrame(simTimeSec, surgeS);

        // Adaptive quality check over the first 90 drawn frames
        if (!useFallback && qualityStage < 2 && rawDtMs > 0) {
          drawnFrameCount++;
          accumulatedFrameMs += rawDtMs;

          const sampleTarget = qualityStage === 0 ? 90 : 60;
          if (drawnFrameCount >= sampleTarget) {
            const avgMs = accumulatedFrameMs / drawnFrameCount;
            drawnFrameCount = 0;
            accumulatedFrameMs = 0;

            if (avgMs > 28) {
              if (qualityStage === 0) {
                // Cut canvas width by 25% (min 220) and set OCT to 3
                qualityStage = 1;
                widthScale = 0.75;
                currentOctaves = 3;
                syncMistCanvasSize();
                if (gl) {
                  const downgraded = createMistProgram(gl, 3);
                  if (downgraded) {
                    mistProg?.dispose();
                    mistProg = downgraded;
                  }
                }
              } else {
                // Still slow: switch to the CSS gradient fallback
                qualityStage = 2;
                activateFallback();
              }
            } else {
              qualityStage = 2;
            }
          }
        }
      }

      rafId = requestAnimationFrame(tick);
    };

    const startLoop = () => {
      if (unmounted || isPaused) return;
      if (rafId === null) {
        lastRafNow = performance.now();
        rafId = requestAnimationFrame(tick);
      }
    };

    const stopLoop = () => {
      if (rafId !== null) {
        cancelAnimationFrame(rafId);
        rafId = null;
      }
    };

    const syncPauseState = () => {
      const nextPaused = !isInView || document.hidden;
      if (nextPaused === isPaused) return;
      isPaused = nextPaused;
      heroEl.classList.toggle('hero-paused', isPaused);
      if (isPaused) {
        stopLoop();
      } else {
        startLoop();
      }
    };

    // Trigger mist puff on photo tap/click (ignored if intro or puff is already running)
    ctrlRef.current.triggerPuff = () => {
      if (
        unmounted ||
        isReducedMotion ||
        isPaused ||
        isIntroActive ||
        isPuffActive
      ) {
        return;
      }
      isPuffActive = true;
      puffElapsedMs = 0;
      startLoop();
    };

    // Decode image, draw 40px sky canvas, and launch intro / loop
    const initAfterDecode = () => {
      if (unmounted) return;
      drawSkyCanvas();
      syncPhotoGeometry();
      applyScrollParallax();

      if (isReducedMotion) {
        renderMistFrame(0, 0);
        return;
      }

      if (!hasPlayedHeroIntro) {
        hasPlayedHeroIntro = true;
        isIntroActive = true;
        introElapsedMs = 0;
      }
      startLoop();
    };

    if (typeof imgEl.decode === 'function') {
      imgEl
        .decode()
        .then(initAfterDecode)
        .catch(() => {
          initAfterDecode();
        });
    } else if (imgEl.complete) {
      initAfterDecode();
    } else {
      imgEl.addEventListener('load', initAfterDecode, { once: true });
    }

    // One passive scroll listener throttled with rAF
    const onScroll = () => {
      scrollDirty = true;
      if (isReducedMotion && !isPaused) {
        applyScrollParallax();
      }
    };
    window.addEventListener('scroll', onScroll, { passive: true });

    // ResizeObserver for photo geometry & mist canvas sizing
    const ro = new ResizeObserver(() => {
      syncPhotoGeometry();
      syncMistCanvasSize();
      scrollDirty = true;
      if (isReducedMotion) {
        renderMistFrame(0, 0);
      }
    });
    ro.observe(heroEl);
    ro.observe(photoStageEl);

    // IntersectionObserver to pause drawing and CSS loops when off-screen
    const io = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          isInView = entry.isIntersecting;
          syncPauseState();
        });
      },
      { threshold: 0.01 }
    );
    io.observe(heroEl);

    // Visibilitychange listener
    const onVisibilityChange = () => {
      syncPauseState();
    };
    document.addEventListener('visibilitychange', onVisibilityChange);

    // Reduced motion listener
    const onReducedMotionChange = (e) => {
      isReducedMotion = Boolean(e.matches);
      heroEl.classList.toggle('hero-reduced-motion', isReducedMotion);
      if (isReducedMotion) {
        isIntroActive = false;
        isPuffActive = false;
        heroEl.classList.remove('is-intro-running');
        heroEl.style.setProperty('--intro', '1');
        if (bloomEl) {
          bloomEl.style.opacity = '';
          bloomEl.style.animation = '';
        }
        renderMistFrame(0, 0);
        stopLoop();
      } else {
        startLoop();
      }
    };
    mqReduced?.addEventListener?.('change', onReducedMotionChange);

    return () => {
      unmounted = true;
      stopLoop();
      window.removeEventListener('scroll', onScroll);
      document.removeEventListener('visibilitychange', onVisibilityChange);
      mqReduced?.removeEventListener?.('change', onReducedMotionChange);
      mistCanvas?.removeEventListener('webglcontextlost', onContextLost);
      ro.disconnect();
      io.disconnect();
      if (mistProg) {
        mistProg.dispose();
        mistProg = null;
      }
      if (gl) {
        try {
          gl.getExtension('WEBGL_lose_context')?.loseContext();
        } catch {
          // Ignore
        }
        gl = null;
      }
    };
  }, []);

  return {
    heroRef,
    skyCanvasRef,
    photoStageRef,
    photoCoordRef,
    imgRef,
    bloomRef,
    mistWrapRef,
    mistCanvasRef,
    fallbackMistRef,
    handlePhotoTap,
  };
}

  const initialDept = searchParams.get('dept') || '';
  const initialDoc = searchParams.get('doc') || '';

  const days = useMemo(() => getNext14Days(), []);

  // Determine starting step if query params pre-populate dept/doc
  const [step, setStep] = useState<1 | 2 | 3 | 4 | 5 | 6>(() => {
    if (initialDept && initialDoc) return 3;
    if (initialDept) return 2;
    return 1;

    morning: SlotItem[];

  }>({ morning: [], evening: [] });
  const [selectedSlot, setSelectedSlot] = useState<string>('');
  const [patientName, setPatientName] = useState<string>('');
  const [patientPhone, setPatientPhone] = useState<string>('');
  const [patientNote, setPatientNote] = useState<string>('');
  const [fieldErrors, setFieldErrors] = useState<{
    name?: string;
    phone?: string;
  }>({});
  const [submitError, setSubmitError] = useState<string>('');
  const [confirmedBooking, setConfirmedBooking] = useState<{
    bookingId: string;
    departmentId: string;
    doctorId: string;
    date: string;
    timeSlot: string;
    name: string;
    phone: string;
    note?: string;
  } | null>(null);

  useEffect(() => {
    let active = true;
    if (!departmentId) {
      setDoctorsForDept([]);
      setDoctorId('');
      return;
    }
    getDoctors(departmentId).then((docs: DoctorItem[]) => {
      if (!active) return;
      setDoctorsForDept(docs);
      const valid = docs.some((d) => d.id === doctorId);
      if (!valid && docs.length > 0) {
        setDoctorId(docs[0].id);
      }
    });
    return () => {
      active = false;
    };
  }, [departmentId, doctorId]);

  // Load slots for selected doctor & date via bookingService.getSlots()
  useEffect(() => {
    let active = true;
    if (!doctorId || !selectedDate) {
      setSlotGroups({ morning: [], evening: [] });

    }
    getSlots(doctorId, selectedDate).then(
      (groups: { morning: SlotItem[]; evening: SlotItem[] }) => {
        if (!active) return;
        setSlotGroups(groups);
        if (selectedSlot) {
          const all = [...groups.morning, ...groups.evening];
          const match = all.find((s) => s.time === selectedSlot);
          if (!match || match.booked) {
            setSelectedSlot('');
          }
        }
      }
    );
    return () => {

const STORAGE_KEY = 'naqsh_hospital_taken_slots_v1';
const BOOKINGS_LIST_KEY = 'naqsh_hospital_bookings_v1';

// Default pre-booked slots per date so users can see greyed-out disabled slots immediately
const DEFAULT_TAKEN_SLOTS = ['10:00 AM', '05:30 PM'];

const MORNING_SLOTS = [
  '09:00 AM',
  '09:30 AM',
  '10:00 AM',
  '10:30 AM',
  '11:00 AM',
  '11:30 AM',
  '12:00 PM',
  '12:30 PM',
];

const EVENING_SLOTS = [
  '04:00 PM',
  '04:30 PM',
  '05:00 PM',
  '05:30 PM',
  '06:00 PM',
  '06:30 PM',
  '07:00 PM',
  '07:30 PM',
];

const inMemoryTakenSlots = new Set();

function formatLocalIsoDate(d) {
  const yyyy = d.getFullYear();
  const mm = String(d.getMonth() + 1).padStart(2, '0');
  const dd = String(d.getDate()).padStart(2, '0');
  return `${yyyy}-${mm}-${dd}`;
}

function parseSlotMinutes(timeStr) {
  const match = String(timeStr).match(/(\d{1,2}):(\d{2})\s*(AM|PM)/i);
  if (!match) return 0;
  let hours = parseInt(match[1], 10);
  const mins = parseInt(match[2], 10);
  const mer = match[3].toUpperCase();
  if (mer === 'PM' && hours < 12) hours += 12;
  if (mer === 'AM' && hours === 12) hours = 0;
  return hours * 60 + mins;
}

function isPastSlotForDate(dateIso, timeStr, now = new Date()) {
  const todayIso = formatLocalIsoDate(now);
  if (dateIso < todayIso) return true;
  if (dateIso > todayIso) return false;
  const nowMinutes = now.getHours() * 60 + now.getMinutes();
  // If current time is already after 7:15 PM, don't block all evening slots on same-day demo view
  if (nowMinutes >= 19 * 60 + 15) return false;
  const slotMinutes = parseSlotMinutes(timeStr);
  return slotMinutes <= nowMinutes;
}

function loadTakenSet() {
  const combined = new Set(inMemoryTakenSlots);
  if (typeof window !== 'undefined' && window.localStorage) {
    try {
      const raw = window.localStorage.getItem(STORAGE_KEY);
      if (raw) {
        const parsed = JSON.parse(raw);
        if (Array.isArray(parsed)) {
          parsed.forEach((key) => combined.add(key));
        }
      }
    } catch {
      // ignore storage errors
    }
  }
  return combined;
}

function saveTakenSlotKey(slotKey, bookingRecord) {
  inMemoryTakenSlots.add(slotKey);
  if (typeof window !== 'undefined' && window.localStorage) {
    try {
      const current = loadTakenSet();
      current.add(slotKey);
      window.localStorage.setItem(
        STORAGE_KEY,
        JSON.stringify(Array.from(current))
      );
      if (bookingRecord) {
        const rawList = window.localStorage.getItem(BOOKINGS_LIST_KEY);
        const list = rawList ? JSON.parse(rawList) : [];
        if (Array.isArray(list)) {
          list.unshift(bookingRecord);
          window.localStorage.setItem(
            BOOKINGS_LIST_KEY,
            JSON.stringify(list.slice(0, 50))
          );
        }
      }
    } catch {
      // ignore storage errors
    }
  }
}

export function buildSlotKey(doctorId, date, time) {
  return `${doctorId}__${date}__${time}`;
}

/**

export async function getDoctors(departmentId) {

  if (!departmentId || departmentId === 'all') {
    return [...all];
  }

/**
 * Returns Morning and Evening slots for a given doctorId and YYYY-MM-DD date.
 * Booked or past slots have `booked: true`.
 * @param {string} doctorId
 * @param {string} date
 */
export async function getSlots(doctorId, date) {
  const takenSet = loadTakenSet();
  const now = new Date();

  const mapSlot = (time, period) => {
    const key = buildSlotKey(doctorId, date, time);
    const isDefaultBooked = DEFAULT_TAKEN_SLOTS.includes(time);
    const isPast = isPastSlotForDate(date, time, now);
    const isBooked = isDefaultBooked || takenSet.has(key) || isPast;
    return {
      time,
      period,
      booked: isBooked,
      isPast,
    };
  };

  return {
    morning: MORNING_SLOTS.map((t) => mapSlot(t, 'morning')),
    evening: EVENING_SLOTS.map((t) => mapSlot(t, 'evening')),
  };
}

/**
 * Creates a booking, marks the slot as taken in memory + localStorage, and returns the created record.
 */
export async function createBooking(data) {
  const {
    departmentId,
    doctorId,
    date,
    timeSlot,
    name,
    phone,
    note = '',
  } = data;
  if (!departmentId || !doctorId || !date || !timeSlot || !name || !phone) {
    throw new Error('Missing required booking fields');
  }

  const todayIso = formatLocalIsoDate(new Date());
  if (date < todayIso) {
    throw new Error('Cannot book an appointment for a past date');
  }

  const slotKey = buildSlotKey(doctorId, date, timeSlot);
  const currentSlots = await getSlots(doctorId, date);
  const allSlots = [...currentSlots.morning, ...currentSlots.evening];
  const targetSlot = allSlots.find((s) => s.time === timeSlot);

  if (targetSlot && targetSlot.booked) {
    throw new Error('Selected time slot is already booked or unavailable');
  }

  const dept =
    hospitalData.departments.find((d) => d.id === departmentId) ||
    hospitalData.departments[0];
  const doc =
    hospitalData.doctors.find((d) => d.id === doctorId) ||
    hospitalData.doctors[0];

  const bookingId = `NQ-${Date.now().toString().slice(-6)}`;

  const record = {
    bookingId,
    departmentId,
    doctorId,
    departmentName: dept.name,
    doctorName: doc.name,
    date,
    timeSlot,
    name: name.trim(),
    phone: phone.trim(),
    note: note.trim(),
    createdAt: new Date().toISOString(),
  };

  saveTakenSlotKey(slotKey, record);

  return record;
}

/**
 * Generates the next 14 days starting from today (or tomorrow if after 7:30 PM).
 * @param {Date} [baseDate]
 */
export function getNext14Days(baseDate = new Date()) {
  const start = new Date(baseDate);
  if (start.getHours() * 60 + start.getMinutes() >= 19 * 60 + 30) {
    start.setDate(start.getDate() + 1);
  }
  const days = [];
  for (let i = 0; i < 14; i++) {
    const d = new Date(start);
    d.setDate(start.getDate() + i);
    const iso = formatLocalIsoDate(d);
    days.push({
      iso,
      dayShortEn: d.toLocaleDateString('en-IN', { weekday: 'short' }),
      dayShortHi: d.toLocaleDateString('hi-IN', { weekday: 'short' }),
      dayNum: String(d.getDate()).padStart(2, '0'),
      monthShortEn: d.toLocaleDateString('en-IN', { month: 'short' }),
      monthShortHi: d.toLocaleDateString('hi-IN', { month: 'short' }),
    });
  }
  return days;
}

export interface ValidationResult {

export function sanitizePhoneDigits(phone: string): string {
  return (phone || '').replace(/\D/g, '');
}

/**
 * Validates Indian mobile numbers of 10 digits (starting 6-9), optionally prefixed with +91.
 * Returns normalized "+91 XXXXX XXXXX" string if valid, or null if invalid.
 */
export function normalizeIndianPhone(phone: string): string | null {
  const trimmed = (phone || '').trim();
  const digits = sanitizePhoneDigits(trimmed);
  if (digits.length === 10 && /^[6-9]\d{9}$/.test(digits)) {
    return `+91 ${digits.slice(0, 5)} ${digits.slice(5)}`;
  }
  if (
    digits.length === 12 &&
    digits.startsWith('91') &&
    (trimmed.startsWith('+91') || trimmed.startsWith('91'))
  ) {
    const core = digits.slice(2);
    if (/^[6-9]\d{9}$/.test(core)) {
      return `+91 ${core.slice(0, 5)} ${core.slice(5)}`;
    }
  }
  return null;
}

export function validateAppointmentForm(
  input: AppointmentFormInput,
  lang: Language = 'en'
): ValidationResult {
  const errors: Partial<Record<keyof AppointmentFormInput, string>> = {};

  if (!input.name || input.name.trim().length < 2) {
    errors.name =
      lang === 'hi'
        ? 'कृपया मरीज का पूरा नाम दर्ज करें'
        : 'Please enter the patient full name';
  }

  const normalizedPhone = normalizeIndianPhone(input.phone || '');
  if (!normalizedPhone) {
    errors.phone =
      lang === 'hi'
        ? 'कृपया 10 अंकों का वैध +91 मोबाइल नंबर दर्ज करें'
        : 'Please enter a valid 10-digit +91 mobile number';
  }

  if (!input.departmentId || input.departmentId.trim() === '') {
    errors.departmentId =
      lang === 'hi' ? 'कृपया विभाग चुनें' : 'Please select a department';
  }

  if (!input.doctorId || input.doctorId.trim() === '') {
    errors.doctorId =
      lang === 'hi' ? 'कृपया डॉक्टर चुनें' : 'Please select a doctor';
  }

  if (!input.date || input.date.trim() === '') {
    errors.date =
      lang === 'hi'
        ? 'कृपया अपॉइंटमेंट की तारीख चुनें'
        : 'Please select an appointment date';
  }

  if (!input.timeSlot || input.timeSlot.trim() === '') {
    errors.timeSlot =
      lang === 'hi'
        ? 'कृपया समय स्लॉट चुनें'
        : 'Please select a preferred time slot';
  }

  return {
    isValid: Object.keys(errors).length === 0,
    errors,
  };
}

export function buildWhatsAppAppointmentUrl(
  whatsappNumber: string,
  formData: AppointmentFormInput & { bookingId?: string },
  departmentName: string,
  doctorName: string
): { whatsappUrl: string; formattedMessage: string } {
  const cleanTargetPhone = sanitizePhoneDigits(whatsappNumber);

function parseDateTimeStrings(dateIso: string, timeSlot: string) {
  const [yyyy, mm, dd] = dateIso.split('-').map(Number);
  const match = timeSlot.match(/(\d{1,2}):(\d{2})\s*(AM|PM)/i);
  let hours = 10;
  let minutes = 0;
  if (match) {
    hours = parseInt(match[1], 10);
    minutes = parseInt(match[2], 10);
    const meridiem = match[3].toUpperCase();
    if (meridiem === 'PM' && hours < 12) hours += 12;
    if (meridiem === 'AM' && hours === 12) hours = 0;
  }
  const pad = (n: number) => String(n).padStart(2, '0');
  const startStr = `${yyyy}${pad(mm)}${pad(dd)}T${pad(hours)}${pad(minutes)}00`;
  const endHours = hours + (minutes + 30 >= 60 ? 1 : 0);
  const endMinutes = (minutes + 30) % 60;
  const endStr = `${yyyy}${pad(mm)}${pad(dd)}T${pad(endHours)}${pad(endMinutes)}00`;
  return { startStr, endStr };
}

  const content = buildIcsCalendarContent(params);
  const blob = new Blob([content], { type: 'text/calendar;charset=utf-8' });

  const link = document.createElement('a');
  link.href = url;
  link.setAttribute('download', `naqsh-appointment-${params.bookingId}.ics`);
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
  setTimeout(() => URL.revokeObjectURL(url), 1000);
}
